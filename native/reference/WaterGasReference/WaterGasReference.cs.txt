// Controlled direct original-UpdateLiquid oracle. No game UI or frame is run.
internal static class WaterGasReference
{
    const string DllHash = "031960F120AC1D59FD8E43F6E1A7369111C629B90C24D52186B691DFE2D8A727";
    const int Width = 5, Height = 5, SimWidth = 7, SimHeight = 7;
    const int Count = Width * Height, SimCount = SimWidth * SimHeight;
    const int Center = 3 * SimWidth + 3;
    static readonly JsonSerializerOptions JsonOptions = new() { WriteIndented = true };
    [UnmanagedFunctionPointer(CallingConvention.Cdecl)] delegate int Callback(int message, IntPtr data);
    [UnmanagedFunctionPointer(CallingConvention.Cdecl)] delegate void Initialize(Callback callback);
    [UnmanagedFunctionPointer(CallingConvention.Cdecl)] delegate void Shutdown();
    [UnmanagedFunctionPointer(CallingConvention.Cdecl)] delegate IntPtr Handle(int id, int bytes, IntPtr data);
    [UnmanagedFunctionPointer(CallingConvention.Cdecl)] delegate void LiquidStep(IntPtr simData, IntPtr events, int simCell);
    [UnmanagedFunctionPointer(CallingConvention.Cdecl)] delegate void CopyFrom(IntPtr destination, IntPtr source);
    sealed record CellPatch(int X,int Y,string Element,float Mass,float Temperature=300f,byte Properties=0);
    sealed record Case(string Name,CellPatch[] Cells,uint Seed=70,int Direction=-1,ushort Tick=0);
    static readonly (string Name, int Offset, int Size)[] Grids = [
        ("elementIdx",0,2),("temperature",0x20,4),("mass",0x40,4),("properties",0x60,1),
        ("insulation",0x80,1),("strengthInfo",0xa0,1),("diseaseIdx",0xc0,1),
        ("diseaseCount",0xe0,4),("diseaseInfestationTickCount",0x100,1),
        ("diseaseGrowthAccumulatedError",0x120,4),("radiation",0x140,4)];
    static readonly string[] EventNames = ["substanceChangeInfo","spawnLiquidInfo","spawnOreInfo","unstableCellInfo",
        "elementChunkMeltedInfo","buildingMeltedInfo","buildingOverheatInfo","buildingNoLongerOverheatedInfo",
        "cellMeltedInfo","callbackInfo","worldDamageInfo","massConsumedCallbacks","radiationConsumedCallbacks",
        "massEmittedCallbacks","diseaseConsumedCallbacks","spawnFXInfo","componentStateChangedMessages",
        "digInfo","backwallElementChangedInfo","backwallShouldTransitionInfo"];
    static string Hash(byte[] bytes) => Convert.ToHexString(SHA256.HashData(bytes)).ToLowerInvariant();
    static string Arg(string[] args, string name) {
        int at = Array.IndexOf(args,name);
        return at >= 0 && at+1 < args.Length ? args[at+1] : throw new ArgumentException("Missing " + name);
    }
    static uint Bits(float value) => BitConverter.SingleToUInt32Bits(value);
    static float Float(uint bits) => BitConverter.UInt32BitsToSingle(bits);
    static IEnumerable<Case> Cases() {
        CellPatch C(int x,int y,float m=500f)=>new(x,y,"water",m);
        CellPatch G(int x,int y,string gas="oxygen",float m=1f,float t=300f,byte p=0)=>new(x,y,gas,m,t,p);
        CellPatch S(int x,int y)=>new(x,y,"solid",9999f,0f);
        foreach(string gas in new[]{"oxygen","co2"}) {
            yield return new("down-"+gas+"-open",[C(3,3),G(3,2,gas)]);
            yield return new("down-"+gas+"-corner",[C(3,3),G(3,2,gas),S(2,2)]);
            for(ushort tick=0;tick<4;tick++) {
                yield return new("side-"+gas+"-tick"+tick,[C(3,3),S(3,2),S(4,3),G(2,3,gas)],Tick:tick);
                yield return new("side-"+gas+"-merge-tick"+tick,[C(3,3),S(3,2),S(4,3),G(2,3,gas),G(2,4,gas,2f,310f)],Tick:tick);
            }
            yield return new("side-"+gas+"-blocked",[C(3,3),S(3,2),S(4,3),G(2,3,gas),S(2,2),S(1,3),S(2,4)]);
            yield return new("side-"+gas+"-zero",[C(3,3),S(3,2),S(4,3),G(2,3,gas,0)]);
            yield return new("side-"+gas+"-diagonal-left",[C(3,3),S(3,2),S(4,3),G(2,3,gas),S(2,2),S(1,3),S(2,4),G(1,4,gas,2f,310f)]);
            yield return new("side-"+gas+"-diagonal-right",[C(3,3),S(3,2),S(4,3),G(2,3,gas),S(2,2),S(1,3),S(2,4),G(3,4,gas,2f,310f)]);
            yield return new("side-"+gas+"-diagonal-vacuum",[C(3,3),S(3,2),S(4,3),G(2,3,gas),S(2,2),S(1,3),S(2,4)]);
            yield return new("above-"+gas+"-open",[C(3,3,2000f),S(3,2),S(2,3),S(4,3),G(3,4,gas)]);
            yield return new("diagonal-"+gas+"-right",[C(3,3),C(3,2,1000f),S(2,3),S(4,3),S(3,4),G(4,2,gas)]);
            yield return new("diagonal-"+gas+"-left",[C(3,3),C(3,2,1000f),S(2,3),S(4,3),S(3,4),G(2,2,gas)]);
            yield return new("diagonal-"+gas+"-both",[C(3,3),C(3,2,1000f),S(2,3),S(4,3),S(3,4),G(2,2,gas),G(4,2,gas)]);
            yield return new("diagonal-"+gas+"-below-solid",[C(3,3),S(3,2),S(2,3),S(4,3),S(3,4),G(2,2,gas),G(4,2,gas)]);

            // Thresholds and updated-neighbour merges are tested through the
            // original single-cell entry, without a preceding physics frame.
            float[] waterMasses=[.005f,.01f,.02f,1f,500f,2000f,4000f];
            float[] gasMasses=[0f,1e-8f,1f,5f];
            ushort[] ticks=[0,1,2,3,65535];
            for(int w=0;w<waterMasses.Length;w++) {
                for(int g=0;g<gasMasses.Length;g++)
                    yield return new($"matrix-down-{gas}-w{w}-g{g}",
                        [C(3,3,waterMasses[w]),G(3,2,gas,gasMasses[g],315f)]);
                foreach(ushort tick in ticks)
                    yield return new($"matrix-side-{gas}-w{w}-tick{tick}",
                        [C(3,3,waterMasses[w]),S(3,2),S(4,3),G(2,3,gas,1f,315f)],Tick:tick);
            }
            for(int g=0;g<gasMasses.Length;g++) foreach(ushort tick in ticks)
                yield return new($"matrix-side-{gas}-g{g}-tick{tick}",
                    [C(3,3),S(3,2),S(4,3),G(2,3,gas,gasMasses[g],315f)],Tick:tick);
            foreach(float mass in new[]{2000f,4000f}) for(int g=0;g<gasMasses.Length;g++)
                yield return new($"matrix-above-{gas}-w{mass}-g{g}",
                    [C(3,3,mass),S(3,2),S(2,3),S(4,3),G(3,4,gas,gasMasses[g],285f)],Tick:65535);
            for(int g=0;g<gasMasses.Length;g++) foreach(int delta in new[]{-15,15})
                yield return new($"matrix-merge-{gas}-g{g}-dt{delta}",
                    [C(3,3),S(3,2),S(4,3),G(2,3,gas,1f,300f),G(2,4,gas,gasMasses[g],300f+delta)]);
            foreach(ushort tick in ticks) {
                yield return new($"candidates-{gas}-all-gas-tick{tick}",
                    [C(3,3),S(3,2),S(4,3),G(2,3,gas,1f,305f),G(2,2,gas,2f,280f),G(1,3,gas,3f,290f),G(2,4,gas,4f,310f)],Tick:tick);
                yield return new($"candidates-{gas}-up-blocked-tick{tick}",
                    [C(3,3),S(3,2),S(4,3),G(2,3,gas,1f,305f),S(2,4),G(2,2,gas,2f,280f),G(1,3,gas,3f,290f)],Tick:tick);
                yield return new($"candidates-{gas}-sides-blocked-tick{tick}",
                    [C(3,3),S(3,2),S(4,3),G(2,3,gas,1f,305f),S(2,4),S(2,2),G(1,3,gas,3f,290f)],Tick:tick);
                yield return new($"candidates-{gas}-other-gas-tick{tick}",
                    [C(3,3),S(3,2),S(4,3),G(2,3,gas,1f,305f),G(2,4,gas=="oxygen"?"co2":"oxygen",2f,280f),G(1,3,gas,3f,290f)],Tick:tick);
                yield return new($"candidates-{gas}-vacuum-versus-merge-tick{tick}",
                    [C(3,3),S(3,2),S(4,3),G(2,3,gas,1f,305f),G(1,3,gas,3f,290f),G(2,4,gas,4f,310f)],Tick:tick);
            }
        }
    }
    static int Main(string[] args) {
        if(!OperatingSystem.IsWindows() || IntPtr.Size!=8) return 2;
        try {
            if(args.Contains("--list-cases")) { foreach(var item in Cases()) Console.WriteLine(item.Name); return 0; }
            if(args.Contains("--suite")) return Suite(args);
            Case selected=Cases().Single(c=>c.Name==Arg(args,"--case"));
            return Run(args,selected);
        } catch(Exception ex) { Console.Error.WriteLine(ex.GetType().Name+": "+ex.Message); return 1; }
    }
    static int Suite(string[] args) {
        string output=Path.GetFullPath(Arg(args,"--output-dir"));Directory.CreateDirectory(output);
        var results=new List<object>(); int completed=0,failed=0;
        string executable=Environment.ProcessPath ?? throw new InvalidOperationException("Process path unavailable");
        foreach(var item in Cases()) {
            string destination=Path.Combine(output,item.Name);
            var start=new ProcessStartInfo(executable) { UseShellExecute=false,CreateNoWindow=true,
                RedirectStandardOutput=true,RedirectStandardError=true };
            if(Path.GetFileNameWithoutExtension(executable).Equals("dotnet",StringComparison.OrdinalIgnoreCase))
                start.ArgumentList.Add(Assembly.GetExecutingAssembly().Location);
            foreach(string text in new[]{"--game-root",Arg(args,"--game-root"),"--capture-dir",Arg(args,"--capture-dir"),
                "--output-dir",destination,"--case",item.Name}) start.ArgumentList.Add(text);
            using Process child=Process.Start(start)!;
            Task<string> stdout=child.StandardOutput.ReadToEndAsync(),stderr=child.StandardError.ReadToEndAsync();
            if(!child.WaitForExit(20000)) { child.Kill(true); throw new TimeoutException(item.Name+" exceeded 20 seconds"); }
            string log=stdout.GetAwaiter().GetResult(),error=stderr.GetAwaiter().GetResult();
            Directory.CreateDirectory(destination);File.WriteAllText(Path.Combine(destination,"process.log"),log+error);
            if(child.ExitCode!=0) ++failed;
            string reportPath=Path.Combine(destination,"report.json");
            results.Add(new { name=item.Name,exit_code=child.ExitCode,report=reportPath,
                report_sha256=File.Exists(reportPath)?Hash(File.ReadAllBytes(reportPath)):null });
            ++completed;
            if(completed%10==0 || child.ExitCode!=0) Console.WriteLine($"Completed {completed}; failed {failed}; last {item.Name}");
            File.WriteAllText(Path.Combine(output,"suite.json"),JsonSerializer.Serialize(new {
                build="U59-744825-V",oracle="direct RVA 0x498B0 UpdateLiquid, no physics frame",
                completed,failed,cases=results },JsonOptions));
            if(child.ExitCode!=0) return 1;
        }
        Console.WriteLine($"Original direct UpdateLiquid oracle completed {completed} cases; failures {failed}");
        return failed==0?0:1;
    }
    static int Run(string[] args, Case item) {
        string game=Path.GetFullPath(Arg(args,"--game-root")),capture=Path.GetFullPath(Arg(args,"--capture-dir"));
        string output=Path.GetFullPath(Arg(args,"--output-dir"));Directory.CreateDirectory(output);
        string native=Path.Combine(game,"OxygenNotIncluded_Data","Plugins","x86_64","SimDLL.dll");
        byte[] nativeBytes=File.ReadAllBytes(native),elements=File.ReadAllBytes(Path.Combine(capture,"elements.bin")),
            diseases=File.ReadAllBytes(Path.Combine(capture,"diseases.bin"));
        if(!Hash(nativeBytes).Equals(DllHash,StringComparison.OrdinalIgnoreCase)) throw new InvalidDataException("Expected exact Build 744825 DLL");
        int elementCount=BitConverter.ToInt32(elements);
        if(elementCount<=0 || elementCount>65534 || elements.Length<4+164*elementCount ||
            diseases.Length<8 || BitConverter.ToInt32(diseases,4)!=elementCount)
            throw new InvalidDataException("Invalid captured tables");
        var ids=new Dictionary<uint,ushort>();
        for(int index=0;index<elementCount;index++) {
            if(BitConverter.ToUInt16(elements,4+index*164+8)!=index) throw new InvalidDataException("Element index mismatch");
            ids.Add(BitConverter.ToUInt32(elements,4+index*164),checked((ushort)index));
        }
        ushort vacuum=ids[0x2d39bf75u];
        ushort Element(string value) => value switch { "water"=>ids[1836671383u],"solid"=>ids[1838482828u],"vacuum"=>vacuum,"oxygen"=>ids[0xa4e0b340u],"co2"=>ids[0x74dc00efu],_=>throw new InvalidDataException("Unknown element") };
        int centerGame=(Height/2)*Width+Width/2;
        byte[] payload=new byte[20+40*Count];
        using(var stream=new MemoryStream(payload)) using(var writer=new BinaryWriter(stream)) {
            writer.Write(Width);writer.Write(Height);writer.Write(item.Seed);writer.Write(false);writer.Write(true);
            for(int index=0;index<Count;index++) {
                int x=index%Width+1,y=index/Width+1;
                CellPatch? patch=item.Cells.LastOrDefault(p=>p.X==x&&p.Y==y);
                writer.Write(patch==null?vacuum:Element(patch.Element));
                writer.Write((byte)0);writer.Write((byte)255);writer.Write(patch?.Properties??(byte)0);writer.Write(new byte[3]);
                writer.Write(patch?.Temperature??0f);writer.Write(patch?.Mass??0f);
            }
            for(int index=0;index<Count;index++) {writer.Write((byte)255);writer.Write(new byte[3]);writer.Write(0);writer.Write(0f);}
            for(int index=0;index<Count;index++) {writer.Write(vacuum);writer.Write((ushort)0);writer.Write(0f);writer.Write(0f);}
        }
        File.WriteAllBytes(Path.Combine(output,"initialize-from-cells.bin"),payload);
        var stages=new List<string>();var phases=new Dictionary<string,object>();
        var report=new Dictionary<string,object> {
            ["build"]="U59-744825-V",["dll_sha256"]=Hash(nativeBytes),["elements_sha256"]=Hash(elements),
            ["diseases_sha256"]=Hash(diseases),["case"]=item,["width"]=Width,["height"]=Height,
            ["sim_width"]=SimWidth,["sim_height"]=SimHeight,["center_sim_cell"]=Center,["center_game_cell"]=centerGame,
            ["native_entry_rva"]="0x498B0",["new_game_frames_submitted"]=0,
            ["diagnostic_overrides"]="CopyFrom(updated to source) RVA 0x6390 before oracle; seed/direction, center timer 0xA2, center flow[1.25,2.5,3.75,5]",
            ["stages"]=stages,["phases"]=phases };
        void Stage(string text) {stages.Add(text);Console.WriteLine(text);File.WriteAllText(Path.Combine(output,"report.json"),JsonSerializer.Serialize(report,JsonOptions));}
        Environment.SetEnvironmentVariable("PATH",Path.GetDirectoryName(native)+Path.PathSeparator+game+Path.PathSeparator+Environment.GetEnvironmentVariable("PATH"));
        IntPtr library=NativeLibrary.Load(native);
        var initialize=Marshal.GetDelegateForFunctionPointer<Initialize>(NativeLibrary.GetExport(library,"SIM_Initialize"));
        var shutdown=Marshal.GetDelegateForFunctionPointer<Shutdown>(NativeLibrary.GetExport(library,"SIM_Shutdown"));
        var handle=Marshal.GetDelegateForFunctionPointer<Handle>(NativeLibrary.GetExport(library,"SIM_HandleMessage"));
        var post=Marshal.GetDelegateForFunctionPointer<LiquidStep>(library+0x498b0);
        var copyFrom=Marshal.GetDelegateForFunctionPointer<CopyFrom>(library+0x6390);
        Callback callback=(id,data)=>{Stage("Native callback id="+id);return 0;};
        IntPtr Send(int id,byte[] bytes) {
            IntPtr ptr=bytes.Length==0?IntPtr.Zero:Marshal.AllocHGlobal(bytes.Length);
            try {if(bytes.Length>0)Marshal.Copy(bytes,0,ptr,bytes.Length);return handle(id,bytes.Length,ptr);}
            finally {if(ptr!=IntPtr.Zero)Marshal.FreeHGlobal(ptr);}
        }
        bool active=false;
        try {
            Stage("Initialize: calling");initialize(callback);active=true;Stage("Initialize: returned");
            Send(1108437482,elements);Stage("Elements: returned");Send(825301935,diseases);Stage("Diseases: returned");
            Send(2062421945,payload);Stage("InitializeFromCells: returned");
            IntPtr update=Send(-931446686,[]);if(update==IntPtr.Zero)throw new InvalidDataException("Start returned null");
            Stage("Start: returned; no frame queued");
            IntPtr sim=Marshal.ReadIntPtr(library,0x115640);
            if(sim==IntPtr.Zero||Marshal.ReadInt32(sim)!=SimWidth||Marshal.ReadInt32(sim,4)!=SimHeight)
                throw new InvalidDataException("Internal SimData dimensions mismatch");
            report["observed_headless"]=Marshal.ReadByte(sim,0x859);
            report["observed_radiation"]=Marshal.ReadByte(sim,0x85a);
            if(Marshal.ReadByte(sim,0x859)!=1 || Marshal.ReadByte(sim,0x85a)!=0)
                throw new InvalidDataException("Native initialization flags mismatch");
            IntPtr soa=Marshal.ReadIntPtr(sim,0x28),events=Marshal.ReadIntPtr(sim,0xa8),flow=Marshal.ReadIntPtr(sim,0x68),timers=Marshal.ReadIntPtr(sim,0x70);
            if(soa==IntPtr.Zero||events==IntPtr.Zero||flow==IntPtr.Zero||timers==IntPtr.Zero)
                throw new InvalidDataException("Required native pointers are null");
            IntPtr sourceSoa=Marshal.ReadIntPtr(sim,0x20);
            Stage("Diagnostic CopyFrom(updated -> source): calling");copyFrom(sourceSoa,soa);Stage("Diagnostic CopyFrom: returned");
            Marshal.WriteInt16(sim,0xa0,unchecked((short)item.Tick));Marshal.WriteInt32(sim,0x0c,unchecked((int)item.Seed));Marshal.WriteInt32(sim,0x14,item.Direction);
            Marshal.Copy(new float[]{1.25f,2.5f,3.75f,5f},0,flow+Center*16,4);Marshal.WriteByte(timers,Center,0xa2);
            void Snapshot(string phase) {
                var arrays=new Dictionary<string,object>();
                byte[] Copy(string name,IntPtr ptr,int length) {
                    byte[] raw=new byte[length];if(length>0)Marshal.Copy(ptr,raw,0,length);
                    File.WriteAllBytes(Path.Combine(output,phase+"."+name+".bin"),raw);
                    arrays[name]=new{bytes=length,sha256=Hash(raw)};return raw;
                }
                foreach(var view in new[]{("source",sourceSoa),("updated",soa)}) foreach(var field in Grids) {
                    IntPtr begin=Marshal.ReadIntPtr(view.Item2,field.Offset+8),end=Marshal.ReadIntPtr(view.Item2,field.Offset+16);
                    if(end-begin!=SimCount*field.Size)throw new InvalidDataException("Vector size mismatch: "+field.Name);
                    Copy(view.Item1+"."+field.Name,begin,SimCount*field.Size);
                }
                Copy("flow",flow,SimCount*16);Copy("timers",timers,SimCount);
                Copy("accumulatedFlow",Marshal.ReadIntPtr(sim,0x60),SimCount*4);
                var eventVectors=new Dictionary<string,object>();
                for(int index=0;index<EventNames.Length;index++) {
                    int offset=index*0x20;IntPtr begin=Marshal.ReadIntPtr(events,offset+8),end=Marshal.ReadIntPtr(events,offset+16);
                    long bytes=end-begin;if(bytes<0||bytes>SimCount*1024L)throw new InvalidDataException("Event vector bounds: "+EventNames[index]);
                    byte[] raw=Copy("events."+EventNames[index],begin,checked((int)bytes));
                    eventVectors[EventNames[index]]=new{bytes,hex=Convert.ToHexString(raw)};
                }
                byte[] ev=File.ReadAllBytes(Path.Combine(output,phase+".events.substanceChangeInfo.bin"));
                if(ev.Length%8!=0)throw new InvalidDataException("Unexpected SubstanceChangeInfo size");
                var changes=new List<object>();
                for(int at=0;at<ev.Length;at+=8)changes.Add(new{cell=BitConverter.ToInt32(ev,at),oldElement=BitConverter.ToUInt16(ev,at+4),newElement=BitConverter.ToUInt16(ev,at+6)});
                phases[phase]=new{seed=unchecked((uint)Marshal.ReadInt32(sim,0xc)),displacementDirection=Marshal.ReadInt32(sim,0x14),
                    iterateDirection=Marshal.ReadInt32(sim,0x10),tick=unchecked((ushort)Marshal.ReadInt16(sim,0xa0)),arrays,event_vectors=eventVectors,substance_changes=changes};
                Stage(phase+": copied 11 source and 11 updated grids, flow, timers, accumulatedFlow, 20 event vectors");
            }
            Snapshot("before");Stage("UpdateLiquid: calling");post(sim,events,Center);Stage("UpdateLiquid: returned");Snapshot("after");
            report["complete"]=true;Stage("Oracle complete");
        } finally {
            if(active) {shutdown();Stage("Shutdown: returned");}GC.KeepAlive(callback);NativeLibrary.Free(library);
        }
        return 0;
    }
}

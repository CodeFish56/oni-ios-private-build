            if(region.Length!=4||region[0]<0||region[1]<0||region[2]<region[0]||region[3]<region[1]||region[2]>width||region[3]>height)
                throw new ArgumentException("Invalid active frame region");
            report["frame_region"]=region;
            byte[] visible=new byte[count]; Array.Fill(visible,(byte)255);
            for(int index=0;index<stepSeconds.Length;index++) {
                string phase="frame"+(index+1).ToString("D3");
                byte[] frame=new byte[28]; BitConverter.GetBytes(stepSeconds[index]).CopyTo(frame,0);
                for(int i=0;i<4;i++) BitConverter.GetBytes(region[i]).CopyTo(frame,4+i*4);
                BitConverter.GetBytes(1f).CopyTo(frame,20);
                File.WriteAllBytes(Path.Combine(output,phase+".input.bin"),frame);
                Send(-775326397,frame);
                IntPtr update=IntPtr.Zero;
                for(int attempt=0;attempt<100;attempt++) {
                    update=Send(1078620451,visible);
                    if(update!=IntPtr.Zero&&Marshal.ReadInt32(update)>0) break;
                    Thread.Sleep(10);
                }
                if(update==IntPtr.Zero||Marshal.ReadInt32(update)!=1) throw new InvalidDataException("Frame did not finish exactly once: "+phase);
                Inspect(phase,update);
                IntPtr frameSave=beginSave(out int frameSaveLength,0,0);
                if(frameSave==IntPtr.Zero||frameSaveLength<29||frameSaveLength>100_000_000) throw new InvalidDataException("Invalid frame save");
                byte[] frameSaved=new byte[frameSaveLength]; Marshal.Copy(frameSave,frameSaved,0,frameSaved.Length); endSave();
                File.WriteAllBytes(Path.Combine(output,phase+".save.bin"),frameSaved);
            }
        }
        Stage("SIM_BeginSave: calling");
        IntPtr saved=beginSave(out int savedLength,0,0);
        if(saved==IntPtr.Zero||savedLength<29||savedLength>100_000_000) throw new InvalidDataException("Invalid BeginSave output");
        byte[] savedBytes=new byte[savedLength]; Marshal.Copy(saved,savedBytes,0,savedLength); endSave();
        File.WriteAllBytes(Path.Combine(output,"original-save.bin"),savedBytes);
        report["original_save"]=new { length=savedLength,sha256=Hash(savedBytes),header_hex=Convert.ToHexString(savedBytes.AsSpan(0,29)),
            version=U32(savedBytes,8),width=I32(savedBytes,12),height=I32(savedBytes,16),x=I32(savedBytes,20),y=I32(savedBytes,24),flag=savedBytes[28] };
        Stage("SIM_BeginSave / SIM_EndSave: returned");
        Stage("SIM_Shutdown: calling"); shutdown(); Stage("SIM_Shutdown: returned");
        GC.KeepAlive(callback); NativeLibrary.Free(library);
        return 0;
    }

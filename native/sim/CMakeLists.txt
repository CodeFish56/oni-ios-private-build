cmake_minimum_required(VERSION 3.16)
project(OniSimCore LANGUAGES CXX)
set(ONI_SIM_ABI_DIR "${CMAKE_CURRENT_SOURCE_DIR}/../NativeABI" CACHE PATH "Directory with the verified native ABI headers")
if(NOT EXISTS "${ONI_SIM_ABI_DIR}/GameDataUpdate.h" AND EXISTS "${CMAKE_CURRENT_SOURCE_DIR}/../abi/GameDataUpdate.h")
    set(ONI_SIM_ABI_DIR "${CMAKE_CURRENT_SOURCE_DIR}/../abi")
endif()
set(ONI_SIM_SOURCES SimCore.cpp SimCoreExports.cpp FrameCellValues.cpp FrameStepSchedule.cpp
    SaveWorldV15.cpp PressureFlux.cpp GasPressure.cpp GasPostProcess.cpp
    StabilityTimers.cpp LiquidFlow.cpp)
add_library(OniSimCore STATIC ${ONI_SIM_SOURCES})
add_library(OniSimCoreShared SHARED ${ONI_SIM_SOURCES})
foreach(target OniSimCore OniSimCoreShared)
    target_compile_features(${target} PUBLIC cxx_std_17)
    target_include_directories(${target} PUBLIC ${CMAKE_CURRENT_SOURCE_DIR} ${ONI_SIM_ABI_DIR})
    if(MSVC)
        target_compile_options(${target} PRIVATE /W4 /WX /EHsc /fp:strict)
    else()
        target_compile_options(${target} PRIVATE -Wall -Wextra -Werror -ffp-contract=off)
    endif()
endforeach()
